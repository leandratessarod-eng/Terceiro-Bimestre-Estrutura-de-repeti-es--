programa 
{
  funcao inicio() 
  {
    inteiro numero 
    inteiro fatorial = 1 
    
    escreva("Digite um numero inteiro: ") 
    leia(numero)

    para (inteiro i = numero; i >= 1; i--)
    { 
      fatorial = fatorial * i 
    }

      escreva("O fatorial de ", numero, " eh: ", fatorial)
      
    
  }
}
