    inteiro menor

      escreva("Quantos numeros voce vai digitar? ")
      leia(n)

      escreva("Digite um numero: ")
      leia(numero)

      menor = numero
      maior = numero
      soma = numero

      para (inteiro i = 2; i <= n; i++)

            menor = numero
          }

          {
            maior = numero
          }

          escreva("\nMenor valor: ", menor)
          escreva("\nMaior valor: ", maior)
          escreva("\nSoma dos valores: ", soma)
       }
   }
